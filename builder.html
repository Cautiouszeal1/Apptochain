10contract DeFiPool is ReentrancyGuard, Ownable {
11 IERC20 public token;
12 uint256 public totalLiquidity;
13 uint256 public FEE = 30; // 0.3%
14 mapping(address => uint256) public shares;
15
16 event LiquidityAdded(address indexed user, uint256 amount, uint256 minted);
17 event LiquidityRemoved(address indexed user, uint256 shares);
18
19 function addLiquidity(uint256 amount) external nonReentrant {
20 require(amount > 0, "Amount must be > 0");
21 uint256 fee = (amount * FEE) / 10000;
22 uint256 net = amount - fee;
23 token.transferFrom(msg.sender, address(this), amount);
24 uint256 minted = totalLiquidity == 0 ? net : (net * totalLiquidity);
25 shares[msg.sender] += minted;
26 totalLiquidity += minted;
27 emit LiquidityAdded(msg.sender, net, minted);
28 }
29
30 function removeLiquidity(uint256 shareAmount) external nonReentrant {
31 require(shares[msg.sender] >= shareAmount, "Insufficient shares");
32 shares[msg.sender] -= shareAmount;
33 totalLiquidity -= shareAmount;
34 emit LiquidityRemoved(msg.sender, shareAmount);
35 }
36
37 function setFee(uint256 _fee) external onlyOwner {
38 require(_fee <= 1000, "Fee too high"); // max 10%
39 FEE = _fee;
40 }
